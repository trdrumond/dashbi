🧩 IMPLEMENTAÇÃO TÉCNICA EM PHP
1️⃣ Instalar SDK OpenAI

Via composer:

composer require openai-php/client
2️⃣ Classe IAService
use OpenAI;

class IAService {

    private $client;

    public function __construct()
    {
        $this->client = OpenAI::client('SUA_API_KEY');
    }

    public function gerarResumoExecutivo(array $dados)
    {
        $prompt = "
        Você é um analista executivo.
        Gere um resumo profissional baseado nos dados:
        " . json_encode($dados);

        $response = $this->client->chat()->create([
            'model' => 'gpt-4o-mini',
            'messages' => [
                ['role' => 'user', 'content' => $prompt]
            ],
            'temperature' => 0.4
        ]);

        return $response->choices[0]->message->content;
    }
}
